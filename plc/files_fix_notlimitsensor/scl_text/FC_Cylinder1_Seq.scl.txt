FUNCTION FC_Cylinder1_Seq : Void
{ S7_Optimized_Access := 'TRUE' }
VERSION : 0.1
// ------------------------------------------------------------
// FC_Cylinder1_Seq (v2 - Timer based, không dùng LS1)
// - Trình tự điều khiển xilanh 1 dùng TON timer thay LS1.
// - State:
//   0 = idle
//   1 = extend: Cylinder1=TRUE, chờ CYL_TIME giây
//   2 = retract: Cylinder1=FALSE, chờ 1 scan → DonePulse
//
// Thời gian cố định: CYL_TIME = T#2S
// Timer instance (TON) phải là VAR trong FB_Auto_Main và truyền VAR_IN_OUT.
// ------------------------------------------------------------

VAR_INPUT
    StartAction : BOOL;
END_VAR

VAR_IN_OUT
    SeqState    : INT;      // 0=idle, 1=extend, 2=retract
    CylTimer    : TON;      // Timer instance — phải là VAR trong FB_Auto_Main
END_VAR

VAR_OUTPUT
    Cylinder1Cmd : BOOL;
    Busy         : BOOL;
    DonePulse    : BOOL;
END_VAR

VAR CONSTANT
    CYL_TIME : TIME := T#2S;
END_VAR

BEGIN
    DonePulse := FALSE;

    CASE SeqState OF
        0: // idle
            Cylinder1Cmd := FALSE;
            Busy         := FALSE;
            CylTimer(IN := FALSE, PT := CYL_TIME); // reset timer
            IF StartAction THEN
                SeqState := 1;
            END_IF;

        1: // extend — giữ Cylinder1=TRUE, chờ timer
            Cylinder1Cmd := TRUE;
            Busy         := TRUE;
            CylTimer(IN := TRUE, PT := CYL_TIME);
            IF CylTimer.Q THEN
                SeqState := 2;
            END_IF;

        2: // retract — tắt Cylinder1, báo Done 1 scan
            Cylinder1Cmd := FALSE;
            Busy         := FALSE;
            CylTimer(IN := FALSE, PT := CYL_TIME); // reset timer
            DonePulse    := TRUE;
            SeqState     := 0;

        ELSE
            SeqState     := 0;
            Cylinder1Cmd := FALSE;
            Busy         := FALSE;
            CylTimer(IN := FALSE, PT := CYL_TIME);
    END_CASE;
END_FUNCTION
