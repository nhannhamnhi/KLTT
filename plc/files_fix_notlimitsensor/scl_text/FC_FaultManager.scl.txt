FUNCTION FC_FaultManager : Void
{ S7_Optimized_Access := 'TRUE' }
VERSION : 0.1
// ------------------------------------------------------------
// FC_FaultManager (v2 - bỏ errLS1Timeout, errLS2Timeout)
// - Gom các lỗi con và xuất ra 2 tín hiệu tổng:
//   - ErrorActive: có lỗi nghiêm trọng cần dừng an toàn
//   - ErrorCode: mã lỗi để debug nhanh trên HMI/Watch table
//
// Mã lỗi:
//   101 = AI timeout (chưa implement timer, giữ chỗ)
//   301 = FIFO overflow
//   302 = FIFO underflow
//   401 = Kết quả AI không hợp lệ (ngoài 0..4)
// ------------------------------------------------------------

VAR_INPUT
    errOverflow      : BOOL;
    errUnderflow     : BOOL;
    errInvalidResult : BOOL;
    errAITimeout     : BOOL;
END_VAR

VAR_OUTPUT
    ErrorActive : BOOL;
    ErrorCode   : INT;
END_VAR

BEGIN
    ErrorActive := FALSE;
    ErrorCode   := 0;

    IF errOverflow THEN
        ErrorActive := TRUE;
        ErrorCode   := 301;
    ELSIF errUnderflow THEN
        ErrorActive := TRUE;
        ErrorCode   := 302;
    ELSIF errInvalidResult THEN
        ErrorActive := TRUE;
        ErrorCode   := 401;
    ELSIF errAITimeout THEN
        ErrorActive := TRUE;
        ErrorCode   := 101;
    END_IF;
END_FUNCTION
